-- Tạo bảng articles lưu trữ bài viết
create table if not exists public.articles (
  id bigint primary key,
  title text not null,
  category text not null,
  image text,
  date text,
  readTime text default '5 phút',
  author text,
  authorRole text,
  excerpt text,
  content text,
  status text default 'published',
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- Bật RLS và cấp quyền đọc/ghi công khai
alter table public.articles enable row level security;
create policy "Cho phep doc cong khai" on public.articles for select using (true);
create policy "Cho phep them bai viet" on public.articles for insert with check (true);
create policy "Cho phep sua bai viet" on public.articles for update using (true);
create policy "Cho phep xoa bai viet" on public.articles for delete using (true);
